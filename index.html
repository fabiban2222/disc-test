        function auswerten() {
          // Antworten einsammeln
          const antworten = {};
          for (let i = 1; i <= 12; i++) {
            const select = document.getElementById('frage' + i);
            antworten['frage' + i] = select.value;
          }

          // Jetzt auswerten
          const profil = berechneProfil(antworten);

          // Ergebnis anzeigen
          const ergebnisDiv = document.getElementById('ergebnis');
          ergebnisDiv.innerHTML = `
            <h2>Dein DISC-Profil</h2>
            <p>Dominanz (D): ${profil.D}</p>
            <p>Initiative (I): ${profil.I}</p>
            <p>Stetigkeit (S): ${profil.S}</p>
            <p>Gewissenhaftigkeit (C): ${profil.C}</p>
            <p><strong>Haupttyp:</strong> ${profil.haupttyp}</p>
          `;
        }


function berechneProfil(antworten) {
  // Mapping: frage -> { a: "D", b: "S", c: "C", d: "I" } usw.
  const mapping = {
    frage1: { a: "D", b: "S", c: "C", d: "I" },
    frage2: { a: "D", b: "S", c: "C", d: "I" },
    frage3: { a: "D", b: "S", c: "C", d: "I" },
    frage4: { a: "D", b: "S", c: "C", d: "I" },
    frage5: { a: "D", b: "S", c: "C", d: "I" },
    frage6: { a: "D", b: "S", c: "C", d: "I" },
    frage7: { a: "D", b: "S", c: "C", d: "I" },
    frage8: { a: "D", b: "S", c: "C", d: "I" },
    frage9: { a: "D", b: "S", c: "C", d: "I" },
    frage10: { a: "D", b: "S", c: "C", d: "I" },
    frage11: { a: "D", b: "S", c: "C", d: "I" },
    frage12: { a: "D", b: "S", c: "C", d: "I" }
  };

  // Zähler für jeden Typ
  const punkte = { D: 0, I: 0, S: 0, C: 0 };

  for (let frage in antworten) {
    const auswahl = antworten[frage];       // a/b/c/d
    const typ = mapping[frage][auswahl];    // D/I/S/C
    punkte[typ]++;
  }

  // Haupttyp bestimmen
  let haupttyp = "D";
  let max = punkte.D;
  for (let typ of ["I", "S", "C"]) {
    if (punkte[typ] > max) {
      max = punkte[typ];
      haupttyp = typ;
    }
  }

  return {
    D: punkte.D,
    I: punkte.I,
    S: punkte.S,
    C: punkte.C,
    haupttyp
  };
}
